Reusable DSPy signature skeletons · input → output

class ReportedCheck(dspy.Signature):
    message: str = dspy.InputField()
    preceding_context: str = dspy.InputField()
    action: str = dspy.OutputField()
    object: str = dspy.OutputField()
    result: str = dspy.OutputField()

class RequestedReview(dspy.Signature):
    message: str = dspy.InputField()
    preceding_context: str = dspy.InputField()
    artifact: str = dspy.OutputField()
    requested_action: str = dspy.OutputField()
